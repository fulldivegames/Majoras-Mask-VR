cmake_minimum_required(VERSION 3.26)
project(MMVRRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(MMVR_CORE_ONLY "Build portable shared code without a platform runtime" OFF)
add_library(mmvr-core INTERFACE)
if(MMVR_LOCAL_TEST_TOOLS)
    target_compile_definitions(mmvr-core INTERFACE MMVR_LOCAL_TEST_TOOLS)
endif()
target_include_directories(mmvr-core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MMVR_CORE_ONLY)
    find_path(MMVR_OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
    target_include_directories(mmvr-core INTERFACE "${MMVR_OPENXR_INCLUDE_DIR}")
else()
    if(NOT TARGET OpenXR::headers)
        find_package(OpenXR CONFIG REQUIRED)
    endif()
    target_link_libraries(mmvr-core INTERFACE OpenXR::headers)
endif()
add_executable(mmvr-core-checks core_tests.cpp)
if(NOT CMAKE_CROSSCOMPILING)
    # Deliberately exclude the private-tools macro: verify public search visibility.
    add_executable(mmvr-menu-search-public-checks menu_search_public_tests.cpp)
    add_test(NAME mmvr-menu-search-public-checks COMMAND mmvr-menu-search-public-checks)
    add_executable(mmvr-prop-contact-checks "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prop-contact-benchmark.cpp")
    target_link_libraries(mmvr-prop-contact-checks PRIVATE mmvr-core)
    add_executable(mmvr-physical-spin-checks physical_spin_volume_test.cpp)
    target_link_libraries(mmvr-physical-spin-checks PRIVATE mmvr-core)
    add_executable(mmvr-periodic-report-checks "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/periodic-report-check.cpp")
    target_compile_definitions(mmvr-periodic-report-checks PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(mmvr-periodic-report-checks PRIVATE mmvr-core)
    add_test(NAME mmvr-periodic-report-checks COMMAND mmvr-periodic-report-checks
        "${CMAKE_CURRENT_BINARY_DIR}/periodic-report-artifacts")
    add_executable(mmvr-bottle-gesture-checks bottle_gesture_test.cpp)
    target_link_libraries(mmvr-bottle-gesture-checks PRIVATE mmvr-core)
    add_test(NAME mmvr-bottle-gesture-checks COMMAND mmvr-bottle-gesture-checks)
endif()
target_link_libraries(mmvr-core-checks PRIVATE mmvr-core)
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_test(NAME mmvr-core-checks COMMAND mmvr-core-checks)
    add_test(NAME mmvr-physical-spin-checks COMMAND mmvr-physical-spin-checks)
endif()
if(MMVR_CORE_ONLY)
    return()
endif()
if(NOT WIN32 AND NOT ANDROID)
    message(FATAL_ERROR "MMVR requires Windows or Android")
endif()
add_library(mmvr-runtime STATIC runtime.cpp)
target_include_directories(mmvr-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mmvr-runtime PUBLIC mmvr-core)
if(ANDROID)
    target_compile_definitions(mmvr-runtime PUBLIC XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
    target_link_libraries(mmvr-runtime PRIVATE mmvr-core OpenXR::openxr_loader OpenXR::headers SDL2::SDL2 EGL GLESv3 android log)
else()
target_compile_definitions(mmvr-runtime PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_compile_options(mmvr-runtime PRIVATE /W4 /EHsc)
target_link_libraries(mmvr-runtime PRIVATE mmvr-core OpenXR::openxr_loader OpenXR::headers d3d11 dxgi)
endif()
option(MMVR_BUILD_TESTS "Build standalone runtime checks" OFF)
if(MMVR_BUILD_TESTS)
  enable_testing()
  add_executable(mmvr-desktop-mirror-check desktop_mirror_check.cpp)
  target_compile_definitions(mmvr-desktop-mirror-check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(mmvr-desktop-mirror-check PRIVATE mmvr-core OpenXR::headers d3d11 dxgi)
  add_test(NAME mmvr-desktop-mirror-check COMMAND mmvr-desktop-mirror-check)
  add_executable(mmvr-runtime-probe runtime_probe.cpp)
  target_link_libraries(mmvr-runtime-probe PRIVATE mmvr-core OpenXR::openxr_loader)
  add_executable(mmvr-runtime-tests tests.cpp)
  target_compile_definitions(mmvr-runtime-tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
  target_link_libraries(mmvr-runtime-tests PRIVATE mmvr-runtime OpenXR::headers d3d11 dxgi)
  add_test(NAME mmvr-runtime-checks COMMAND mmvr-runtime-tests)
  add_test(NAME mmvr-missing-runtime COMMAND mmvr-runtime-tests --missing-runtime)
endif()

set(MMVR_IMGUI_DIR "" CACHE PATH "Optional local ImGui source for actual menu offscreen verification")
if(MMVR_BUILD_TESTS AND MMVR_IMGUI_DIR)
  add_executable(mmvr-ui-preview ui_preview.cpp
    "${MMVR_IMGUI_DIR}/imgui.cpp" "${MMVR_IMGUI_DIR}/imgui_draw.cpp"
    "${MMVR_IMGUI_DIR}/imgui_tables.cpp" "${MMVR_IMGUI_DIR}/imgui_widgets.cpp"
    "${MMVR_IMGUI_DIR}/backends/imgui_impl_dx11.cpp")
  target_include_directories(mmvr-ui-preview PRIVATE "${MMVR_IMGUI_DIR}")
  target_link_libraries(mmvr-ui-preview PRIVATE mmvr-runtime OpenXR::headers d3d11 dxgi d3dcompiler)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ui-artifacts")
  add_test(NAME mmvr-ui-composition COMMAND mmvr-ui-preview "${CMAKE_CURRENT_BINARY_DIR}/ui-artifacts")
endif()
